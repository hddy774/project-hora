using System.Text.Json;
using Chronicles.Game;
using Chronicles.Game.Cultivation;

var policy = new CultivationPolicy();
var actor = new CultivationActor { PersonId = 1, BodyId = 2, SoulId = 3, Realm = Realm.Mortal, Health = 100, Status = LifeStatus.Alive };
var state = CultivationProfiles.ForNewBody(2, 3, "independent-review-declared-body") with
{ EntryWork = policy.RequiredWork(Realm.Mortal), GrowthBasis = new(1, 0, 0, policy.RequiredWork(Realm.Mortal)) };
var path = new KnowledgeEvidence("path.1.0", "valid-qi-edition", 7, 5000, 5000, 0, true, false, "actual independently learned Qi source r7");
var rows = new List<object>();
void Check(string name, KnowledgeEvidence? entry, CultivationActor? chosenActor = null)
{
    var context = new CultivationContext { Month = 2, LawRevision = 1, Policy = policy,
        Site = new() { PlaceId = 0, HasUseRight = true, HasReservedSeat = true, SeatCapacity = 2,
            Available = Elements.Uniform(1000000), FormationOperational = true, FormationQuality = 5000,
            CelestialWindow = true, ObservedRealmConditions = true, RealPassageAccessible = true },
        Knowledge = entry is null ? [path] : [entry, path] };
    var actual = chosenActor ?? actor;
    var result = CultivationEngine.Execute(state, actual, new PrepareCultivation("review-" + name, state.Revision, "path.1.0"), context, 9999);
    var potential = CultivationQueries.Potential(state, actual, context);
    rows.Add(new { name, result.Accepted, blockers = result.Blockers.Select(b => b.Code).ToArray(),
        work = result.State.Attempt?.Knowledge.WorkId, source = result.State.Attempt?.Knowledge.Source,
        revision = result.State.Attempt?.Knowledge.Revision, quality = result.State.Attempt?.FormationQuality,
        supply_debit = result.SupplyDebit.Total, state_unchanged_on_reject = result.Accepted || JsonSerializer.Serialize(state) == JsonSerializer.Serialize(result.State),
        ui_route_blockers = potential.KnownNextRoutes.Single(r => r.Path == "path.1.0").Missing.Select(b => b.Code).ToArray() });
}
Check("valid_qi_only", null);
Check("valid_qi_plus_low_entry", new("entry", "entry-low", 2, 500, 0, 0, true, false, "actual entry source 5 percent"));
Check("valid_qi_plus_refuted_entry", new("entry", "entry-refuted", 3, 5000, 0, 8000, true, false, "actual refuted entry source"));
Check("valid_qi_plus_unavailable_entry", new("entry", "entry-unavailable", 4, 5000, 0, 0, false, false, "entry no longer consultable"));
var weakActor = actor with { InnateRoots = Elements.Uniform(1), Circulation = 0, Body = 0, Mind = 3000 };
Check("weak_body_valid_qi_only", null, weakActor);
Check("weak_body_valid_qi_plus_minimum_entry", new("entry", "entry-minimum", 5, 1000, 0, 0, true, false, "entry minimum source"), weakActor);
Console.WriteLine(JsonSerializer.Serialize(new { reviewed_commit = "728d2a6751d22d2f8fa94894d0c8144600217e1f", kind = "small-pure-domain-review", entry_target = state.EntryWork, rows }));
