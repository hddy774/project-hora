string Description(string key, string? parameter = null)
{
    if (!comments.TryGetValue(key, out var item)) return "";
    var value = parameter == null ? item.Element("summary") : item.Elements("param").FirstOrDefault(p => (string?)p.Attribute("name") == parameter);
    return string.Join(" ", (value?.Value ?? "").Split((char[]?)null, StringSplitOptions.RemoveEmptyEntries));
}
